import ollama
def stream_response(prompt):
    Convo=[]
    Convo.append({"role":"user","content":prompt})
    stream_Chank=ollama.chat(
        model="llama3.2",
        messages=Convo,
        stream=True
    )
    response_data=""
    for Data in stream_Chank:
        response_data+=Data["message"]["content"]
    return response_data
def get_prompt():
    prompt=input("Enter your prompt: ")
    print(type(prompt))
    return prompt
while True:
    user_prompt=get_prompt()
    if user_prompt.lower()=='exit':
        print("existing the program.")
        break
    response=stream_response(user_prompt)
    print("response:",response)
